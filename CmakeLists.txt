cmake_minimum_required(VERSION 3.10)
project(Chip8Emu)

# Set C++ standard just tells cmake to compile using c++17
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Source files
set(SOURCES
    main.cpp
    Chip8.cpp
    Platform.cpp
)

# Add executable
add_executable(Chip8Emu ${SOURCES})
# -----------------------------
# SDL3 paths
# -----------------------------
set(SDL3_ROOT "C:/Users/User/Downloads/SDL3-devel-3.4.0-mingw/SDL3-3.4.0/x86_64-w64-mingw32")
set(SDL3_INCLUDE_DIR "${SDL3_ROOT}/include")          # include folder
set(SDL3_LIBRARY "${SDL3_ROOT}/lib/libSDL3.dll.a")   # main SDL3 library

# Include headers
include_directories(${SDL3_INCLUDE_DIR})

# Link libraries
target_link_libraries(Chip8Emu PRIVATE ${SDL3_LIBRARY})

# Optional: copy SDL3.dll to build folder after build (so your program runs)
add_custom_command(TARGET Chip8Emu POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${SDL3_ROOT}/bin/SDL3.dll"
    $<TARGET_FILE_DIR:Chip8Emu>
)
